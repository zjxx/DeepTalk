## 🔑 核心代码示例

### 1. 前端API调用示例
```typescript
// 社区API - 获取帖子列表
export const getPosts = (page: number, size: number) => {
  return request.get<PageResult<Post>>('/api/community/posts', {
    params: { page, size }
  });
};

// 点赞帖子
export const likePost = (postId: number) => {
  return request.post(`/api/community/posts/${postId}/like`);
};
```

### 2. 后端API实现示例
```java
// 社区控制器 - 获取帖子列表
@GetMapping("/posts")
public ResponseEntity<PageResult<Post>> getPosts(
    @RequestParam(defaultValue = "1") int page,
    @RequestParam(defaultValue = "10") int size
) {
    PageResult<Post> result = communityService.getPosts(page, size);
    return ResponseEntity.ok(result);
}

// 点赞功能
@PostMapping("/posts/{id}/like")
public ResponseEntity<Void> likePost(@PathVariable Long id) {
    communityService.likePost(id);
    return ResponseEntity.ok().build();
}
```
